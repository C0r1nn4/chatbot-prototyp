// ===================================================================== // MSK-Plus Prototyp – Edge Function "msk-plus" // Aufgaben: Sitzung starten, Klicks protokollieren, KI-Chat (Campus-KI / // KI:connect.nrw) weiterleiten und protokollieren, Daten für die Auswertung. // // Secrets (Supabase → Edge Functions → Secrets): // KI_API_KEY Schlüssel für KI:connect / Campus-KI (Pflicht) // ADMIN_TOKEN langes Passwort für die Auswertungsseite (Pflicht) // KI_URL optional, Standard: KI:connect-Endpunkt // KI_MODELL optional, Standard: gpt-oss-120b // ZUGANG_MODUS optional: "liste" (nur angelegte Codes, Standard) oder "offen" // MAX_FRAGEN_PRO_STUNDE optional, Standard 40 (je Teilnahmecode) // ERLAUBTE_ORIGINS optional, z. B. "https://www.meine-seite.de" (kommagetrennt) // SUPABASE_URL und SUPABASE_SERVICE_ROLE_KEY stellt Supabase automatisch bereit. // // Wichtig: In den Einstellungen der Function "Verify JWT" AUSSCHALTEN – // die Seiten senden keinen Supabase-Schlüssel, geprüft wird der Teilnahmecode. // ===================================================================== import { createClient } from "npm:@supabase/supabase-js@2"; declare const EdgeRuntime: { waitUntil(p: Promise): void } | undefined; const KI_URL = Deno.env.get("KI_URL") || "https://chat.kiconnect.nrw/api/v1/chat/completions"; const MODELL = Deno.env.get("KI_MODELL") || "gpt-oss-120b"; const KI_KEY = Deno.env.get("KI_API_KEY") || ""; const ADMIN_TOKEN = Deno.env.get("ADMIN_TOKEN") || ""; const ZUGANG = (Deno.env.get("ZUGANG_MODUS") || "liste").toLowerCase(); const MAX_PRO_STUNDE = Number(Deno.env.get("MAX_FRAGEN_PRO_STUNDE") || "40"); const ORIGINS = (Deno.env.get("ERLAUBTE_ORIGINS") || "*").split(",").map((s) => s.trim()).filter(Boolean); const MAX_KONTEXT_ZEICHEN = 400_000; // Schutz gegen übergroße Anfragen const MAX_EREIGNISSE = 300; const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!, { auth: { persistSession: false }, }); function cors(req: Request): Record { const o = req.headers.get("origin") || ""; const allow = ORIGINS.includes("*") ? "*" : ORIGINS.includes(o) ? o : ORIGINS[0] || ""; return { "Access-Control-Allow-Origin": allow, "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Vary": "Origin", }; } const json = (req: Request, body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { ...cors(req), "Content-Type": "application/json" } }); const normCode = (c: unknown) => String(c ?? "").trim().toUpperCase().replace(/\s+/g, "").slice(0, 40); // Sitzung gehört zum Code? (kleiner Cache je Instanz) const sitzungCache = new Map(); async function pruefeSitzung(code: string, sitzung: string): Promise { if (!code || !sitzung) return false; if (sitzungCache.get(sitzung) === code) return true; const { data } = await sb.from("sitzungen").select("code").eq("id", sitzung).maybeSingle(); if (data && data.code === code) { sitzungCache.set(sitzung, code); return true; } return false; } // ---------------------------------------------------------------- start async function start(req: Request, b: any) { const code = normCode(b.code); if (!code) return json(req, { fehler: "Bitte einen Teilnahmecode eingeben." }, 400); const { data: t } = await sb.from("teilnehmer").select("code, aktiv").eq("code", code).maybeSingle(); if (!t) { if (ZUGANG !== "offen") return json(req, { fehler: "Dieser Teilnahmecode ist nicht bekannt." }, 403); const { error } = await sb.from("teilnehmer").insert({ code, notiz: "selbst angelegt" }); if (error) return json(req, { fehler: "Code konnte nicht angelegt werden." }, 500); } else if (!t.aktiv) { return json(req, { fehler: "Dieser Teilnahmecode ist nicht mehr aktiv." }, 403); } const { data, error } = await sb.from("sitzungen").insert({ code, einwilligung_version: String(b.einwilligung_version ?? "").slice(0, 40), einwilligung_zeit: b.einwilligung_zeit ?? new Date().toISOString(), browser: String(b.browser ?? "").slice(0, 300), bildschirm: String(b.bildschirm ?? "").slice(0, 40), }).select("id").single(); if (error) return json(req, { fehler: "Sitzung konnte nicht gestartet werden." }, 500); sitzungCache.set(data.id, code); return json(req, { sitzung: data.id, code }); } // ---------------------------------------------------------------- log async function log(req: Request, b: any) { const code = normCode(b.code); if (!(await pruefeSitzung(code, b.sitzung))) return json(req, { fehler: "Sitzung unbekannt." }, 403); const ev = Array.isArray(b.ereignisse) ? b.ereignisse.slice(0, MAX_EREIGNISSE) : []; if (!ev.length) return json(req, { ok: true, n: 0 }); const rows = ev.map((e: any) => ({ sitzung_id: b.sitzung, code, seq: Number.isFinite(e.seq) ? e.seq : null, zeit: e.zeit && !isNaN(Date.parse(e.zeit)) ? e.zeit : new Date().toISOString(), typ: String(e.typ ?? "unbekannt").slice(0, 40), ziel: e.ziel == null ? null : String(e.ziel).slice(0, 200), text: e.text == null ? null : String(e.text).slice(0, 2000), details: e.details ?? null, ansicht: e.ansicht ?? null, })); const { error } = await sb.from("ereignisse").insert(rows); if (error) return json(req, { fehler: "Speichern fehlgeschlagen." }, 500); await sb.from("sitzungen").update({ letzte_aktivitaet: new Date().toISOString() }).eq("id", b.sitzung); return json(req, { ok: true, n: rows.length }); } // ---------------------------------------------------------------- chat async function chat(req: Request, b: any) { const code = normCode(b.code); if (!(await pruefeSitzung(code, b.sitzung))) return json(req, { fehler: "Sitzung unbekannt. Bitte über die Startseite neu starten." }, 403); if (!KI_KEY) return json(req, { fehler: "KI-Schlüssel fehlt (Secret KI_API_KEY)." }, 500); const msgs = Array.isArray(b.messages) ? b.messages : []; const ok = msgs.length > 0 && msgs.every((m: any) => ["system", "user", "assistant"].includes(m?.role) && typeof m?.content === "string"); if (!ok || msgs[msgs.length - 1].role !== "user") return json(req, { fehler: "Ungültige Anfrage." }, 400); const laenge = msgs.reduce((n: number, m: any) => n + m.content.length, 0); if (laenge > MAX_KONTEXT_ZEICHEN) return json(req, { fehler: "Anfrage zu lang." }, 413); const seit = new Date(Date.now() - 3600_000).toISOString(); const { count } = await sb.from("chat_nachrichten").select("id", { count: "exact", head: true }) .eq("code", code).eq("rolle", "user").gte("zeit", seit); if ((count ?? 0) >= MAX_PRO_STUNDE) { return json(req, { fehler: "Das Fragen-Limit für diese Stunde ist erreicht. Bitte später weiterfragen." }, 429); } const frage = msgs[msgs.length - 1].content; await sb.from("chat_nachrichten").insert({ sitzung_id: b.sitzung, code, rolle: "user", inhalt: frage.slice(0, 20000), ansicht: b.ansicht ?? null, modell: MODELL, meta: { kontext_zeichen: laenge, nachrichten: msgs.length, client_id: b.client_id ?? null }, }); const t0 = Date.now(); const speichereAntwort = (inhalt: string, rolle: "assistant" | "fehler", meta: Record = {}) => sb.from("chat_nachrichten").insert({ sitzung_id: b.sitzung, code, rolle, inhalt, ansicht: b.ansicht ?? null, modell: MODELL, dauer_ms: Date.now() - t0, meta: { ...meta, client_id: b.client_id ?? null }, }); let up: Response; try { up = await fetch(KI_URL, { method: "POST", headers: { "Authorization": `Bearer ${KI_KEY}`, "Content-Type": "application/json" }, body: JSON.stringify({ model: MODELL, messages: msgs, stream: true }), }); } catch (e) { await speichereAntwort(String(e), "fehler"); return json(req, { fehler: "Der KI-Dienst ist nicht erreichbar." }, 502); } if (!up.ok || !up.body) { const t = await up.text().catch(() => ""); await speichereAntwort(`HTTP ${up.status}: ${t.slice(0, 2000)}`, "fehler"); return json(req, { fehler: `Der KI-Dienst meldet einen Fehler (${up.status}).` }, 502); } const ct = up.headers.get("content-type") || ""; if (!ct.includes("text/event-stream")) { // Dienst antwortet ohne Streaming const j = await up.json().catch(() => null); const text = j?.choices?.[0]?.message?.content ?? ""; await speichereAntwort(text, text ? "assistant" : "fehler", { usage: j?.usage ?? null, stream: false }); return json(req, { text }); } // Streaming: ein Zweig geht an den Browser, der andere wird mitgelesen und gespeichert – // auch wenn die Lehrkraft die Seite vorher schließt. const [zumBrowser, zumSpeichern] = up.body.tee(); const mitlesen = (async () => { const dec = new TextDecoder(); const reader = zumSpeichern.getReader(); let buf = "", text = "", usage: unknown = null; for (;;) { const { done, value } = await reader.read(); if (done) break; buf += dec.decode(value, { stream: true }); let i; while ((i = buf.indexOf("\n")) >= 0) { const line = buf.slice(0, i).trim(); buf = buf.slice(i + 1); if (!line.startsWith("data:")) continue; const d = line.slice(5).trim(); if (!d || d === "[DONE]") continue; try { const j = JSON.parse(d); const c = j?.choices?.[0]?.delta?.content; if (typeof c === "string") text += c; if (j?.usage) usage = j.usage; } catch { /* unvollständige Zeile ignorieren */ } } } await speichereAntwort(text, text ? "assistant" : "fehler", { usage, stream: true }); })().catch((e) => speichereAntwort(String(e), "fehler")); if (typeof EdgeRuntime !== "undefined") EdgeRuntime.waitUntil(mitlesen); return new Response(zumBrowser, { headers: { ...cors(req), "Content-Type": "text/event-stream; charset=utf-8", "Cache-Control": "no-cache" }, }); } // ---------------------------------------------------------------- admin (Auswertung) async function alle(tabelle: string, spalten: string, filter?: { code?: string }) { const out: unknown[] = []; for (let von = 0; ; von += 1000) { let q = sb.from(tabelle).select(spalten).order("id", { ascending: true }).range(von, von + 999); if (filter?.code) q = q.eq("code", filter.code); const { data, error } = await q; if (error) throw error; out.push(...(data ?? [])); if (!data || data.length < 1000) break; } return out; } async function admin(req: Request, b: any) { if (!ADMIN_TOKEN || b.token !== ADMIN_TOKEN) return json(req, { fehler: "Kein Zugriff." }, 401); const filter = b.code ? { code: normCode(b.code) } : undefined; try { const { data: teilnehmer } = await sb.from("teilnehmer").select("*").order("erstellt"); let sq = sb.from("sitzungen").select("*").order("start"); if (filter?.code) sq = sq.eq("code", filter.code); const { data: sitzungen } = await sq; const ereignisse = await alle("ereignisse", "id,sitzung_id,code,seq,zeit,server_zeit,typ,ziel,text,details,ansicht", filter); const chat = await alle("chat_nachrichten", "id,sitzung_id,code,zeit,rolle,inhalt,ansicht,modell,dauer_ms,meta", filter); return json(req, { teilnehmer, sitzungen, ereignisse, chat, modell: MODELL }); } catch (e) { return json(req, { fehler: String(e) }, 500); } } // ---------------------------------------------------------------- Modelle (Diagnose) // Aufruf im Browser: …/functions/v1/msk-plus?modelle → zeigt die Modellnamen, die der Schlüssel nutzen darf async function modelle(req: Request) { if (!KI_KEY) return json(req, { fehler: "KI-Schlüssel fehlt (Secret KI_API_KEY)." }, 500); const url = KI_URL.replace(/\/chat\/completions\/?$/, "") + "/models"; try { const r = await fetch(url, { headers: { "Authorization": `Bearer ${KI_KEY}` } }); const j = await r.json().catch(() => null); const ids = Array.isArray(j?.data) ? j.data.map((m: any) => m.id) : null; return json(req, { eingestellt: MODELL, verfuegbar: ids ?? j, status: r.status }); } catch (e) { return json(req, { fehler: String(e) }, 502); } } // ---------------------------------------------------------------- Router Deno.serve(async (req) => { if (req.method === "OPTIONS") return new Response("ok", { headers: cors(req) }); if (req.method === "GET" && new URL(req.url).searchParams.has("modelle")) return modelle(req); if (req.method !== "POST") return json(req, { fehler: "Nur POST." }, 405); let b: any; try { b = JSON.parse(await req.text()); // auch text/plain (sendBeacon, ohne Preflight) } catch { return json(req, { fehler: "Ungültiges JSON." }, 400); } switch (b?.aktion) { case "start": return start(req, b); case "log": return log(req, b); case "chat": return chat(req, b); case "admin": return admin(req, b); case "ping": return json(req, { ok: true, modell: MODELL, zugang: ZUGANG }); default: return json(req, { fehler: "Unbekannte Aktion." }, 400); } });
